zoom 1
center @cnote+150,60
expect sel = none
dbl @cnote+150,90
wait 600
expect sel.around = " number.[]"
drag @cnote+30,60 @cnote+150,60 400
wait 300
expect sel.around = "Th[e vocabulary maps ]each wor"
key esc
wait 300
dbl @cnote+150,90
wait 600
drag @cnote+30,75 @cnote+150,75 400
wait 300
